vvv_select_regiment = {
    scope = country
    saved_scopes = { vvv_battle vvv_regiment }
    is_shown = { always = yes }
    is_valid = { vvv_can_select_regiment = yes }
    effect = {
        if = {
            limit = { vvv_can_select_regiment = yes }
            set_variable = {
                name = vvv_selected_regiment
                value = scope:vvv_regiment
            }
        }
    }
}

# An instantaneous change avoids leaving a combat modifier on a marching army.
# EU5 sub_unit scopes cannot store variables. Keep a timed country map keyed
# by the exact regiment; changing selection does not reset its recovery.
vvv_rally_regiment = {
    scope = country
    saved_scopes = { vvv_battle vvv_regiment }
    is_shown = { always = yes }
    is_valid = { vvv_can_rally_regiment = yes }
    effect = {
        if = {
            limit = { vvv_can_rally_regiment = yes }
            scope:vvv_regiment = {
                add_subunit_morale = 0.10
            }
            add_to_variable_map = {
                name = vvv_rally_cooldowns
                key = scope:vvv_regiment
                value = yes
                days = 7
            }
        }
    }
}
